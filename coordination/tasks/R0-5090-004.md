```bash
(

test "$(hostname)" = "DESKTOP-B5PNBNC"
conda create -n longpibench-r0 python=3.11 pip -y
conda run --no-capture-output -n longpibench-r0 python -m pip install 'torch==2.7.1' --index-url https://download.pytorch.org/whl/cu128

conda run --no-capture-output -n longpibench-r0 python -m pip check
)
```
